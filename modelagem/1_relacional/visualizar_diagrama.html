Estrutura Curricular TADS 2019 — UFRN. Mapeamento normalizado com resolução de matrizes via MATRIZ_COMPONENTE e pré-requisitos via auto-relacionamento PRE_REQUISITO.

    CURSO ||--o{ ESTRUTURA_CURRICULAR : "1:N - oferta"
    ESTRUTURA_CURRICULAR ||--o{ MATRIZ_COMPONENTE : "1:N - estrutura"
    DISCIPLINA ||--o{ MATRIZ_COMPONENTE : "1:N - integra"
    DISCIPLINA ||--o{ PRE_REQUISITO : "1:N - exige (alvo)"
    DISCIPLINA ||--o{ PRE_REQUISITO : "1:N - e_requisito_de"

    CURSO {
        varchar(10) codigo PK "Identificador único"
        varchar(20) sigla UK "Sigla usual (TADS)"
        varchar(150) nome "Nome oficial completo"
        varchar(50) nivel "Graduação"
        varchar(120) unidade "EAJ - UFRN"
        varchar(30) modalidade "Presencial"
    }

    ESTRUTURA_CURRICULAR {
        varchar(20) codigo_matriz PK "Ex: TADS2019"
        varchar(10) curso_codigo FK "FK para CURSO"
        int ano_versao "2019"
        int ch_total_minima "2100"
        int ch_obrigatoria "1680"
        int ch_optativa "240"
        int ch_complementar "180"
        int duracao_semestres "6"
        boolean ativa "true"
    }

    DISCIPLINA {
        varchar(10) codigo PK "Ex: TAD0102"
        varchar(150) nome "Nome da disciplina"
        int ch "Carga horária total (90h)"
        int creditos "Créditos acadêmicos (6)"
        varchar(30) departamento "EAJ"
        text ementa "Conteúdo programático"
    }

    MATRIZ_COMPONENTE {
        int id PK "ID sequencial"
        varchar(20) estrutura_codigo FK
        varchar(10) disciplina_codigo FK
        int periodo_ideal "1 a 6"
        varchar(30) tipo_vinculo "OBRIGATORIA"
    }

    PRE_REQUISITO {
        int id PK "ID sequencial"
        varchar(10) disciplina_alvo_codigo FK
        varchar(10) disciplina_requisito_codigo FK
        varchar(5) operador_logico "E ou OU"
    }
